ด้านที่ 1: องค์ประกอบและมาตรฐานการเรียนรู้/ตัวชี้วัด

เต็ม 20

1.1 มีองค์ประกอบของแผนการจัดการเรียนรู้ครบถ้วนตามรูปแบบที่โรงเรียนมวกเหล็กวิทยากำหนด

1.2 วิเคราะห์มาตรฐานการเรียนรู้ ตัวชี้วัด/ผลการเรียนรู้ ถูกต้องและสอดคล้องกับหลักสูตรสถานศึกษา

1.3 กำหนดสาระสำคัญ/ความคิดรวบยอด ถูกต้อง ครอบคลุมเนื้อหาสาระที่จัดการเรียนรู้

1.4 กำหนดจุดประสงค์การเรียนรู้ชัดเจน ครอบคลุมด้านความรู้ (K) ทักษะ (P) และคุณลักษณะ (A)

ด้านที่ 2: การออกแบบกิจกรรมการเรียนรู้เชิงรุก Active Learning

เต็ม 25

2.1 ออกแบบกิจกรรมการเรียนรู้โดยเน้นผู้เรียนเป็นสำคัญ (Active Learning) เหมาะสมกับวัยและศักยภาพ

2.2 กิจกรรมการเรียนรู้มีลำดับขั้นตอนชัดเจน (ขั้นนำ ขั้นจัดกิจกรรมการเรียนรู้ ขั้นสรุป)

2.3 เปิดโอกาสให้ผู้เรียนได้ลงมือปฏิบัติจริง คิดวิเคราะห์ แก้ปัญหา และทำงานร่วมกันเป็นทีม

2.4 บูรณาการทักษะในศตวรรษที่ 21 หลักปรัชญาของเศรษฐกิจพอเพียง หรือบริบทท้องถิ่นมวกเหล็ก

2.5 กำหนดเวลาในการจัดกิจกรรมการเรียนรู้แต่ละขั้นตอนได้อย่างเหมาะสมและปฏิบัติได้จริง

ด้านที่ 3: สื่อ นวัตกรรม และแหล่งเรียนรู้ดิจิทัล

เต็ม 15

3.1 เลือกใช้สื่อการเรียนรู้ นวัตกรรม และเทคโนโลยีดิจิทัลที่หลากหลาย ทันสมัย และน่าสนใจ

3.2 สื่อ ใบความรู้ และใบงานมีความเหมาะสมกับเนื้อหา ระดับชั้น และความแตกต่างระหว่างบุคคล

3.3 ส่งเสริมให้ผู้เรียนใช้แหล่งเรียนรู้ทั้งภายในและภายนอกห้องเรียนอย่างสร้างสรรค์

ด้านที่ 4: การวัดและประเมินผลตามสภาพจริง

เต็ม 20

4.1 กำหนดวิธีการวัดและประเมินผลสอดคล้องกับจุดประสงค์การเรียนรู้ครบทั้งด้าน K-P-A

4.2 ใช้เครื่องมือวัดและประเมินผลที่หลากหลายตามสภาพจริง (Authentic Assessment)

4.3 กำหนดเกณฑ์การประเมินผล (Rubric Scoring) และเกณฑ์การผ่านอย่างชัดเจนเป็นรูปธรรม

4.4 มีแบบทดสอบ ใบงาน หรือเครื่องมือประเมินผลแนบท้ายแผนการจัดการเรียนรู้อย่างครบถ้วน

ด้านที่ 5: คุณภาพบันทึกหลังแผนและการนำผลไปพัฒนาผู้เรียน

5.2 บันทึกผลการจัดการเรียนรู้สอดคล้องกับจุดประสงค์การเรียนรู้และแสดงผลสัมฤทธิ์ของผู้เรียนชัดเจน

5.3 วิเคราะห์ปัญหา อุปสรรค และข้อค้นพบที่เกิดขึ้นระหว่างการจัดกิจกรรมการเรียนรู้ได้อย่างตรงประเด็น

5.4 เสนอแนวทางการแก้ไขและพัฒนาผู้เรียนเพื่อนำไปปรับปรุงการสอนในครั้งต่อไปอย่างเป็นรูปธรรม

function doGet(e) {
  var planUrl = (e && e.parameter.planUrl) || "";
  var refUrl = (e && e.parameter.refUrl) || "";
  var planInfo = inspectDriveUrl(planUrl);
  var refInfo = inspectDriveUrl(refUrl);

function inspectDriveUrl(url) {
  if (!url) return { exists: false, fileCount: 0, files: [], text: "", reason: "ไม่มีลิงก์" };
  var idMatch = url.match(/[-\w]{25,}/);
  if (!idMatch) return { exists: false, fileCount: 0, files: [], text: "", reason: "ลิงก์ไม่ถูกต้อง" };
  var id = idMatch[0];
  try {
    // Check if Folder
    var folder = DriveApp.getFolderById(id);
    var files = folder.getFiles();
    var names = [], fullText = "";
    while (files.hasNext()) {
      var f = files.next();
      names.push(f.getName());
      fullText += extractFileText(f) + "\n";
    }
    return {
      exists: names.length > 0,
      isFolder: true,
      fileCount: names.length,
      files: names,
      text: fullText.slice(0, 12000),
      reason: names.length > 0 ? "พบไฟล์ในโฟลเดอร์ " + names.length + " ไฟล์" : "โฟลเดอร์ว่างเปล่า (ไม่มีไฟล์ข้างใน)"
    };
  } catch (errFolder) {
    try {
      // Check if Single File
      var file = DriveApp.getFileById(id);
      var txt = extractFileText(file);
      return {
        exists: true,
        isFolder: false,
        fileCount: 1,
        files: [file.getName()],
        text: txt.slice(0, 12000),
        reason: "พบไฟล์ " + file.getName()
      };
    } catch (errFile) {
      return { exists: false, fileCount: 0, files: [], text: "", reason: "ไม่สามารถเข้าถึงไฟล์ได้ (ยังไม่เปิดสิทธิ์แชร์ หรือไฟล์ถูกลบ)" };
    }
  }
}

function extractFileText(file) {
  var mime = file.getMimeType();
  try {
    if (mime === MimeType.GOOGLE_DOCS) {
      return DocumentApp.openById(file.getId()).getBody().getText();
    } else if (mime === MimeType.PLAIN_TEXT) {
      return file.getBlob().getDataAsString();
    } else if (mime === MimeType.PDF || mime.indexOf("word") !== -1) {
      // Convert PDF/DOCX temporarily via Drive API if enabled, or return filename metadata
      return "[ไฟล์เอกสาร: " + file.getName() + " (" + mime + ")]";
    }
    return "[ไฟล์: " + file.getName() + "]";
  } catch (e) {
    return "[ไฟล์: " + file.getName() + "]";
  }
}
