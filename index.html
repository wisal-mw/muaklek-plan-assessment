Phase 2: เชื่อมต่อ Google Apps Script ดึงเนื้อหาจาก Google Drive อัตโนมัติ

1 วางลิงก์ Google Apps Script Web App URL ของโรงเรียน

เมื่อวาง URL ของ Web App ที่ Deploy แล้ว ระบบจะสามารถอ่านเนื้อหาจริงจากไฟล์ Google Docs, PDF, และ Word (.docx) ภายใน Google Drive ของครูทุกคนได้ทันทีโดยไม่ต้องดาวน์โหลดไฟล์ด้วยตนเอง

2 โค้ด Google Apps Script (คัดลอกไปวางใน script.google.com)

  1. เปิด script.google.com ด้วยบัญชี Google โรงเรียนที่เข้าถึงโฟลเดอร์แผนการสอนได้ แล้วสร้างโปรเจกต์ใหม่
  2. ลบโค้ดเดิมออกทั้งหมด แล้ววางโค้ดด้านล่างนี้ลงไป (เปิดใช้บริการเสริม Drive API ในเมนู Services ด้านซ้าย เพื่อให้แปลงไฟล์ PDF/Word เป็นข้อความได้สมบูรณ์)
  3. กดปุ่ม Deploy > New deployment > เลือกประเภทเป็น Web app ตั้งค่า Execute as: Me และ Who has access: Anyone แล้วนำ URL มาวางในช่องด้านบน
/**
 * Google Apps Script Middleware (Phase 2) - โรงเรียนมวกเหล็กวิทยา
 * ดึงเนื้อหาข้อความจาก Google Drive File / Folder URLs (Google Docs, PDF, Word)
 */
function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) || "ping";
  if (action === "ping") {
    return jsonResponse({ status: "ok", school: "โรงเรียนมวกเหล็กวิทยา", version: "Phase2-DriveBridge" });
  }
  if (action === "extract") {
    var planUrl = e.parameter.planUrl || "";
    var refUrl = e.parameter.refUrl || "";
    var planText = planUrl ? extractContentFromDriveUrl(planUrl) : "";
    var refText = refUrl ? extractContentFromDriveUrl(refUrl) : "";
    return jsonResponse({
      status: "ok",
      planText: planText.slice(0, 14000),
      refText: refText.slice(0, 10000)
    });
  }
  return jsonResponse({ status: "error", message: "Unknown action" });
}

function extractContentFromDriveUrl(url) {
  try {
    var folderMatch = url.match(/\/folders\/([a-zA-Z0-9_-]+)/);
    if (folderMatch) {
      var folder = DriveApp.getFolderById(folderMatch[1]);
      var files = folder.getFiles();
      var combined = [];
      var count = 0;
      while (files.hasNext() && count < 3) {
        var f = files.next();
        combined.push("=== ไฟล์: " + f.getName() + " ===\n" + readDriveFileText(f));
        count++;
      }
      return combined.join("\n\n");
    }
    var fileMatch = url.match(/\/d\/([a-zA-Z0-9_-]+)/) || url.match(/[?&]id=([a-zA-Z0-9_-]+)/);
    if (fileMatch) {
      var file = DriveApp.getFileById(fileMatch[1]);
      return "=== ไฟล์: " + file.getName() + " ===\n" + readDriveFileText(file);
    }
    return "";
  } catch (err) {
    return "[ไม่สามารถอ่านไฟล์ได้: " + err.message + "]";
  }
}

function readDriveFileText(file) {
  var mime = file.getMimeType();
  if (mime === MimeType.GOOGLE_DOCS) {
    return DocumentApp.openById(file.getId()).getBody().getText();
  }
  if (mime === MimeType.PLAIN_TEXT) {
    return file.getBlob().getDataAsString("UTF-8");
  }
  // แปลง PDF หรือ Microsoft Word (.docx) เป็น Google Doc ชั่วคราวเพื่อสกัดข้อความ (ต้องเปิด Drive API Service)
  if (typeof Drive !== "undefined" && Drive.Files) {
    var resource = { title: "temp_extract_" + file.getName(), mimeType: MimeType.GOOGLE_DOCS };
    var tempDoc = Drive.Files.copy(resource, file.getId(), { ocr: true, ocrLanguage: "th" });
    var text = DocumentApp.openById(tempDoc.id).getBody().getText();
    DriveApp.getFileById(tempDoc.id).setTrashed(true);
    return text;
  }
  return "[ไฟล์ประเภท " + mime + " กรุณาเปิดใช้ Drive API Service ใน Apps Script]";
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
