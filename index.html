function doGet(e) {
  var planUrl = (e && e.parameter.planUrl) || "";
  var refUrl = (e && e.parameter.refUrl) || "";
  var planInfo = inspectDriveUrl(planUrl);
  var refInfo = inspectDriveUrl(refUrl);
  return ContentService.createTextOutput(JSON.stringify({
    ok: true,
    plan: planInfo,
    reflection: refInfo
  })).setMimeType(ContentService.MimeType.JSON);
}

function inspectDriveUrl(url) {
  if (!url) return { exists: false, fileCount: 0, files: [], text: "", reason: "ไม่มีลิงก์" };
  var idMatch = url.match(/[-\w]{25,}/);
  if (!idMatch) return { exists: false, fileCount: 0, files: [], text: "", reason: "ลิงก์ไม่ถูกต้อง" };
  var id = idMatch[0];
  try {
    // Check if Folder
    var folder = DriveApp.getFolderById(id);
    var files = folder.getFiles();
    var names = [], fullText = "";
    while (files.hasNext()) {
      var f = files.next();
      names.push(f.getName());
      fullText += extractFileText(f) + "\n";
    }
    return {
      exists: names.length > 0,
      isFolder: true,
      fileCount: names.length,
      files: names,
      text: fullText.slice(0, 12000),
      reason: names.length > 0 ? "พบไฟล์ในโฟลเดอร์ " + names.length + " ไฟล์" : "โฟลเดอร์ว่างเปล่า (ไม่มีไฟล์ข้างใน)"
    };
  } catch (errFolder) {
    try {
      // Check if Single File
      var file = DriveApp.getFileById(id);
      var txt = extractFileText(file);
      return {
        exists: true,
        isFolder: false,
        fileCount: 1,
        files: [file.getName()],
        text: txt.slice(0, 12000),
        reason: "พบไฟล์ " + file.getName()
      };
    } catch (errFile) {
      return { exists: false, fileCount: 0, files: [], text: "", reason: "ไม่สามารถเข้าถึงไฟล์ได้ (ยังไม่เปิดสิทธิ์แชร์ หรือไฟล์ถูกลบ)" };
    }
  }
}

function extractFileText(file) {
  var mime = file.getMimeType();
  try {
    if (mime === MimeType.GOOGLE_DOCS) {
      return DocumentApp.openById(file.getId()).getBody().getText();
    } else if (mime === MimeType.PLAIN_TEXT) {
      return file.getBlob().getDataAsString();
    } else if (mime === MimeType.PDF || mime.indexOf("word") !== -1) {
      // Convert PDF/DOCX temporarily via Drive API if enabled, or return filename metadata
      return "[ไฟล์เอกสาร: " + file.getName() + " (" + mime + ")]";
    }
    return "[ไฟล์: " + file.getName() + "]";
  } catch (e) {
    return "[ไฟล์: " + file.getName() + "]";
  }
}
